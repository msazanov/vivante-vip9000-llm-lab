cmake_minimum_required(VERSION 3.18)
project(q1_cpu_operator_runner LANGUAGES C CXX)

set(PRISM_SOURCE "" CACHE PATH "Pinned PrismML source tree")
set(PRISM_BUILD "" CACHE PATH "Accepted PrismML native build tree")
option(BUILD_PINNED_PRISM_FROM_SOURCE "Build Prism ggml from PRISM_SOURCE" OFF)
option(ALLOW_UNPINNED_PRISM "Allow an unpinned Prism source commit (unqualified)" OFF)
set(PRISM_PINNED_SHA "38c66ad0241da4f9fcce541cda8edc219086cec5" CACHE STRING "Required Prism source SHA")

if(NOT PRISM_SOURCE)
    message(FATAL_ERROR "PRISM_SOURCE is required")
endif()
if(NOT IS_ABSOLUTE "${PRISM_SOURCE}")
    message(FATAL_ERROR "PRISM_SOURCE must be an absolute path")
endif()
if(NOT EXISTS "${PRISM_SOURCE}/.git")
    message(FATAL_ERROR "PRISM_SOURCE must be a Git checkout")
endif()
find_package(Git REQUIRED)
execute_process(
    COMMAND "${GIT_EXECUTABLE}" -C "${PRISM_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE PRISM_SOURCE_SHA
    OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE PRISM_SHA_RESULT
    ERROR_VARIABLE PRISM_SHA_ERROR)
if(NOT PRISM_SHA_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not read PRISM_SOURCE Git SHA: ${PRISM_SHA_ERROR}")
endif()
message(STATUS "Prism source SHA: ${PRISM_SOURCE_SHA}")
if(NOT PRISM_SOURCE_SHA STREQUAL PRISM_PINNED_SHA AND NOT ALLOW_UNPINNED_PRISM)
    message(FATAL_ERROR "Prism source SHA ${PRISM_SOURCE_SHA} differs from pinned ${PRISM_PINNED_SHA}; set ALLOW_UNPINNED_PRISM=ON to build unqualified")
endif()

if(BUILD_PINNED_PRISM_FROM_SOURCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    add_subdirectory("${PRISM_SOURCE}/ggml" "${CMAKE_BINARY_DIR}/prism-ggml")
    set(PRISM_INCLUDE_DIRS
        "${PRISM_SOURCE}/ggml/include"
        "${PRISM_SOURCE}/ggml/src")
else()
    if(NOT PRISM_BUILD)
        message(FATAL_ERROR "PRISM_BUILD is required when BUILD_PINNED_PRISM_FROM_SOURCE=OFF")
    endif()
    if(NOT IS_ABSOLUTE "${PRISM_BUILD}")
        message(FATAL_ERROR "PRISM_BUILD must be an absolute path")
    endif()
    if(NOT EXISTS "${PRISM_BUILD}/CMakeCache.txt")
        message(FATAL_ERROR "PRISM_BUILD must contain CMakeCache.txt")
    endif()
    file(STRINGS "${PRISM_BUILD}/CMakeCache.txt" PRISM_HOME_LINE REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
    string(REGEX REPLACE "^CMAKE_HOME_DIRECTORY:INTERNAL=" "" PRISM_HOME "${PRISM_HOME_LINE}")
    file(REAL_PATH "${PRISM_SOURCE}" PRISM_SOURCE_REAL)
    file(REAL_PATH "${PRISM_HOME}" PRISM_HOME_REAL)
    if(NOT PRISM_HOME_REAL STREQUAL PRISM_SOURCE_REAL)
        message(FATAL_ERROR "PRISM_BUILD CMAKE_HOME_DIRECTORY ${PRISM_HOME_REAL} does not match PRISM_SOURCE ${PRISM_SOURCE_REAL}")
    endif()
    file(STRINGS "${PRISM_BUILD}/CMakeCache.txt" PRISM_GGML_SOURCE_LINE REGEX "^ggml_SOURCE_DIR:STATIC=")
    string(REGEX REPLACE "^ggml_SOURCE_DIR:STATIC=" "" PRISM_GGML_SOURCE "${PRISM_GGML_SOURCE_LINE}")
    file(REAL_PATH "${PRISM_SOURCE}/ggml" PRISM_GGML_REAL)
    file(REAL_PATH "${PRISM_GGML_SOURCE}" PRISM_GGML_SOURCE_REAL)
    if(NOT PRISM_GGML_SOURCE_REAL STREQUAL PRISM_GGML_REAL)
        message(FATAL_ERROR "PRISM_BUILD ggml_SOURCE_DIR ${PRISM_GGML_SOURCE_REAL} does not match PRISM_SOURCE/ggml ${PRISM_GGML_REAL}")
    endif()
    set(PRISM_INCLUDE_DIRS
        "${PRISM_SOURCE}/ggml/include"
        "${PRISM_SOURCE}/ggml/src"
        "${PRISM_BUILD}/ggml/src")
    link_directories("${PRISM_BUILD}/bin" "${PRISM_BUILD}/lib")
endif()

add_executable(q1_cpu_operator_runner q1_cpu_operator_runner.cpp)
target_compile_features(q1_cpu_operator_runner PRIVATE cxx_std_17)
target_include_directories(q1_cpu_operator_runner PRIVATE ${PRISM_INCLUDE_DIRS})
if(ALLOW_UNPINNED_PRISM)
    target_compile_definitions(q1_cpu_operator_runner PRIVATE Q1_STRICT_MODE=0)
else()
    target_compile_definitions(q1_cpu_operator_runner PRIVATE Q1_STRICT_MODE=1)
endif()
target_link_libraries(q1_cpu_operator_runner PRIVATE ggml ggml-base ggml-cpu)
if(NOT BUILD_PINNED_PRISM_FROM_SOURCE)
    set_target_properties(q1_cpu_operator_runner PROPERTIES
        BUILD_RPATH "${PRISM_BUILD}/bin;${PRISM_BUILD}/lib")
endif()
